cmake_minimum_required(VERSION 3.28)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/common/bootstrap.cmake")
  include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/common/bootstrap.cmake" NO_POLICY_SCOPE)
  project(${_name} VERSION ${_version} LANGUAGES C CXX)
  include(compilerconfig)
  include(defaults)
  include(helpers)
else()
  project(camera-popout-native VERSION 0.2.0 LANGUAGES C CXX)
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Camera Pop-Out Native requires 64-bit Windows")
endif()
find_package(libobs CONFIG REQUIRED)
find_package(obs-frontend-api CONFIG REQUIRED)
add_library(${PROJECT_NAME} MODULE src/plugin-main.cpp)
target_compile_features(${PROJECT_NAME} PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(${PROJECT_NAME} PRIVATE /utf-8)
endif()
target_compile_definitions(${PROJECT_NAME} PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_link_libraries(${PROJECT_NAME} PRIVATE OBS::libobs OBS::obs-frontend-api user32 gdi32 dwmapi)
set_target_properties(${PROJECT_NAME} PROPERTIES PREFIX "" OUTPUT_NAME "camera-popout-native")
if(COMMAND set_target_properties_plugin)
  set_target_properties_plugin(${PROJECT_NAME} PROPERTIES OUTPUT_NAME "camera-popout-native")
endif()
